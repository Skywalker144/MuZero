cmake_minimum_required(VERSION 3.18)
project(muzero LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
option(MUZERO_WITH_TORCH "Build LibTorch selfplay" ON)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(PROTOCOL_HEADER ${CMAKE_CURRENT_BINARY_DIR}/generated/muzero/protocol.h)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/generated/muzero)
add_custom_command(OUTPUT ${PROTOCOL_HEADER}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/generate_protocol.py
        ${CMAKE_CURRENT_SOURCE_DIR}/../protocol.json ${PROTOCOL_HEADER}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../protocol.json ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/generate_protocol.py
    VERBATIM)
add_library(muzero_core src/search.cpp src/batcher.cpp src/rules.cpp src/random_evaluator.cpp src/record.cpp src/training_targets.cpp src/opening.cpp src/eval_session.cpp ${PROTOCOL_HEADER})
target_include_directories(muzero_core PUBLIC include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(muzero_core PUBLIC Threads::Threads ZLIB::ZLIB)
target_compile_options(muzero_core PRIVATE -Wall -Wextra -Wpedantic)
if(MUZERO_WITH_TORCH)
    find_package(Torch REQUIRED)
    separate_arguments(TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
    target_compile_options(muzero_core PUBLIC ${TORCH_FLAGS})
    add_library(muzero_torch src/torch_backend.cpp)
    target_link_libraries(muzero_torch PUBLIC muzero_core "${TORCH_LIBRARIES}")
    add_executable(muzero_selfplay src/selfplay_main.cpp)
    target_link_libraries(muzero_selfplay PRIVATE muzero_torch)
    add_executable(muzero_benchmark src/benchmark_main.cpp)
    target_link_libraries(muzero_benchmark PRIVATE muzero_torch)
    add_executable(muzero_eval src/eval_main.cpp)
    target_link_libraries(muzero_eval PRIVATE muzero_torch)
endif()
include(CTest)
if(BUILD_TESTING)
    if(MUZERO_WITH_TORCH)
        add_executable(torch_backend_test tests/torch_backend_test.cpp)
        target_link_libraries(torch_backend_test PRIVATE muzero_torch)
    endif()
    add_test(NAME resolve_config COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/../python/muzero/config.py
        --config-dir configs/baseline --native --output ${CMAKE_CURRENT_BINARY_DIR}/test.cfg)
    set_tests_properties(resolve_config PROPERTIES FIXTURES_SETUP resolved_config)
    add_executable(rules_test tests/rules_test.cpp)
    target_link_libraries(rules_test PRIVATE muzero_core)
    add_test(NAME rules COMMAND rules_test)
    add_executable(core_test tests/core_test.cpp)
    target_link_libraries(core_test PRIVATE muzero_core)
    add_test(NAME core COMMAND core_test ${CMAKE_CURRENT_BINARY_DIR}/test.cfg)
    set_tests_properties(core PROPERTIES FIXTURES_REQUIRED resolved_config)
    add_executable(alignment_test tests/alignment_test.cpp)
    target_link_libraries(alignment_test PRIVATE muzero_core)
    add_test(NAME alignment COMMAND alignment_test ${CMAKE_CURRENT_BINARY_DIR}/test.cfg)
    set_tests_properties(alignment PROPERTIES FIXTURES_REQUIRED resolved_config TIMEOUT 30)
    add_executable(search_parallel_test tests/search_parallel_test.cpp)
    target_link_libraries(search_parallel_test PRIVATE muzero_core)
    add_test(NAME search_parallel COMMAND search_parallel_test ${CMAKE_CURRENT_BINARY_DIR}/test.cfg)
    set_tests_properties(search_parallel PROPERTIES FIXTURES_REQUIRED resolved_config TIMEOUT 30)
    add_executable(eval_session_test tests/eval_session_test.cpp)
    target_link_libraries(eval_session_test PRIVATE muzero_core)
    add_test(NAME eval_session COMMAND eval_session_test ${CMAKE_CURRENT_BINARY_DIR}/test.cfg)
    set_tests_properties(eval_session PROPERTIES FIXTURES_REQUIRED resolved_config TIMEOUT 30)
    add_executable(batcher_test tests/batcher_test.cpp)
    target_link_libraries(batcher_test PRIVATE muzero_core)
    add_test(NAME batcher COMMAND batcher_test)
    add_executable(random_evaluator_test tests/random_evaluator_test.cpp)
    target_link_libraries(random_evaluator_test PRIVATE muzero_core)
    add_test(NAME random_evaluator COMMAND random_evaluator_test ${CMAKE_CURRENT_BINARY_DIR}/test.cfg)
    set_tests_properties(random_evaluator PROPERTIES FIXTURES_REQUIRED resolved_config TIMEOUT 30)
    add_executable(opening_test tests/opening_test.cpp)
    target_link_libraries(opening_test PRIVATE muzero_core)
    add_test(NAME opening COMMAND opening_test ${CMAKE_CURRENT_BINARY_DIR}/test.cfg)
    set_tests_properties(opening PROPERTIES FIXTURES_REQUIRED resolved_config TIMEOUT 30)
    set_tests_properties(core batcher rules PROPERTIES TIMEOUT 30)
